```c
/* Opcodes (8-bit) */
#define XPROBE_OPCODE_HALT       0x00
#define XPROBE_OPCODE_LOAD_IMM   0x01
#define XPROBE_OPCODE_LOAD_CTX   0x02
#define XPROBE_OPCODE_LOAD_MAP   0x03
#define XPROBE_OPCODE_STORE_MAP  0x04
#define XPROBE_OPCODE_ADD        0x05
#define XPROBE_OPCODE_SUB        0x06
#define XPROBE_OPCODE_CMP_GT     0x07
#define XPROBE_OPCODE_JMP_IF_NOT 0x08
#define XPROBE_OPCODE_INC_MAP    0x09

/* Context field codes */
#define XPROBE_CTX_USER_DATA     0
#define XPROBE_CTX_TIMESTAMP_NS  1
#define XPROBE_CTX_SCHEMA_ID     2
#define XPROBE_CTX_FLAGS         3
#define XPROBE_CTX_WRITE_POS     4
#define XPROBE_CTX_READ_POS      5
#define XPROBE_CTX_CAPACITY      6
#define XPROBE_CTX_PAYLOAD_LEN   7

/* Instruction encoding macros (32-bit fixed width) */
#define XPROBE_LOAD_IMM(dst, imm) \
  ((uint32_t)(((XPROBE_OPCODE_LOAD_IMM) << 24) | (((dst) & 0x7) << 21) | ((imm) & 0x3FFFF)))

#define XPROBE_LOAD_CTX(dst, field) \
  ((uint32_t)(((XPROBE_OPCODE_LOAD_CTX) << 24) | (((dst) & 0x7) << 21) | ((field) & 0x7)))

#define XPROBE_LOAD_MAP(dst, idx) \
  ((uint32_t)(((XPROBE_OPCODE_LOAD_MAP) << 24) | (((dst) & 0x7) << 21) | ((idx) & 0x3FFFF)))

#define XPROBE_STORE_MAP(src, idx) \
  ((uint32_t)(((XPROBE_OPCODE_STORE_MAP) << 24) | (((src) & 0x7) << 21) | ((idx) & 0x3FFFF)))

#define XPROBE_ADD(dst, src) \
  ((uint32_t)(((XPROBE_OPCODE_ADD) << 24) | (((dst) & 0x7) << 21) | (((src) & 0x7) << 18)))

#define XPROBE_SUB(dst, src) \
  ((uint32_t)(((XPROBE_OPCODE_SUB) << 24) | (((dst) & 0x7) << 21) | (((src) & 0x7) << 18)))

#define XPROBE_CMP_GT(dst, src) \
  ((uint32_t)(((XPROBE_OPCODE_CMP_GT) << 24) | (((dst) & 0x7) << 21) | (((src) & 0x7) << 18)))

#define XPROBE_JMP_IF_NOT(offset) \
  ((uint32_t)(((XPROBE_OPCODE_JMP_IF_NOT) << 24) | ((offset) & 0x3FFFF)))

#define XPROBE_INC_MAP(idx) \
  ((uint32_t)(((XPROBE_OPCODE_INC_MAP) << 24) | ((idx) & 0x3FFFF)))

#define XPROBE_HALT() \
  ((uint32_t)(XPROBE_OPCODE_HALT << 24))

#endif /* XPROBE_INSTRUCTIONS_H */
```

```c
#include <xproc/probe/probe.h>
#include <xproc/probe/probe_instructions.h>

/* Internal: validate bytecode. Returns XPROBE_OK or negative error. */
int xprobe_verify(const uint32_t* instructions, uint32_t count) {
  if (!instructions) return XPROBE_ERR_INVALID_PROG;
  if (count == 0) return XPROBE_ERR_INVALID_PROG;
  if (count > XPROBE_MAX_INSTRUCTIONS) return XPROBE_ERR_INVALID_PROG;

  /* Must end with HALT */
  uint32_t last = instructions[count - 1];
  uint8_t last_op = (uint8_t)(last >> 24);
  if (last_op != XPROBE_OPCODE_HALT) return XPROBE_ERR_INVALID_PROG;

  for (uint32_t i = 0; i < count; i++) {
    uint32_t instr = instructions[i];
    uint8_t opcode = (uint8_t)(instr >> 24);
    uint8_t dst = (uint8_t)((instr >> 21) & 0x7);
    uint8_t src = (uint8_t)((instr >> 18) & 0x7);

    switch (opcode) {
      case XPROBE_OPCODE_HALT:
        break;

      case XPROBE_OPCODE_LOAD_IMM:
        /* dst in [0,7], imm18 is fine */
        if (dst > 7) return XPROBE_ERR_INVALID_PROG;
        break;

      case XPROBE_OPCODE_LOAD_CTX:
        if (dst > 7) return XPROBE_ERR_INVALID_PROG;
        /* field code in low 3 bits */
        if ((instr & 0x7) > 7) return XPROBE_ERR_INVALID_PROG;
        break;

      case XPROBE_OPCODE_LOAD_MAP:
      case XPROBE_OPCODE_STORE_MAP:
      case XPROBE_OPCODE_INC_MAP:
        if (opcode != XPROBE_OPCODE_INC_MAP && dst > 7) return XPROBE_ERR_INVALID_PROG;
        /* map_idx in low 18 bits — checked at exec time against map size */
        break;

      case XPROBE_OPCODE_ADD:
      case XPROBE_OPCODE_SUB:
      case XPROBE_OPCODE_CMP_GT:
        if (dst > 7 || src > 7) return XPROBE_ERR_INVALID_PROG;
        break;

      case XPROBE_OPCODE_JMP_IF_NOT: {
        /* offset must be positive (no backward jumps) and within bounds */
        int32_t offset = (int32_t)(instr & 0x3FFFF);
        if (offset <= 0) return XPROBE_ERR_INVALID_PROG;
        if ((int64_t)i + offset >= (int64_t)count) return XPROBE_ERR_INVALID_PROG;
        break;
      }

      default:
        return XPROBE_ERR_INVALID_PROG;
    }
  }

  return XPROBE_OK;
}
```

```c
#include <xproc/probe/probe.h>
#include <xproc/probe/probe_instructions.h>
#include <stdlib.h>
#include <string.h>

/* ---- Internal structures ---- */

struct xprobe_prog {
  uint32_t*        instructions;
  uint32_t         count;
  xprobe_license   license;
  char*            license_data;  /* owned copy */
};

struct xprobe_map {
  char      name[XPROBE_MAP_NAME_MAX];
  uint32_t  size;
  uint64_t* counters;
};

struct xprobe_handle {
  xprobe_prog* prog;
  xprobe_map*  map;
  xprobe_hook  hook;
};

/* ---- License verification ---- */

static xprobe_license_verify_fn g_verify_fn = NULL;
static void*                    g_verify_user_data = NULL;

void xprobe_set_license_verify(xprobe_license_verify_fn fn, void* user_data) {
  g_verify_fn = fn;
  g_verify_user_data = user_data;
}

static int verify_license(const xprobe_license* lic) {
  if (!g_verify_fn) return XPROBE_OK;  /* no verifier = pass */
  if (!lic || !lic->data || lic->len == 0) return XPROBE_OK;  /* no license = pass (open source) */
  return g_verify_fn(lic, g_verify_user_data);
}

/* ---- Verifier (from probe_verify.c, linked together) ---- */
extern int xprobe_verify(const uint32_t* instructions, uint32_t count);

/* ---- Program lifecycle ---- */

xprobe_prog* xprobe_prog_create(const uint32_t* instructions, uint32_t count,
                                 const xprobe_license* license) {
  int rc = xprobe_verify(instructions, count);
  if (rc != XPROBE_OK) return NULL;

  rc = verify_license(license);
  if (rc != XPROBE_OK) return NULL;

  xprobe_prog* p = (xprobe_prog*)calloc(1, sizeof(xprobe_prog));
  if (!p) return NULL;

  p->instructions = (uint32_t*)malloc(count * sizeof(uint32_t));
  if (!p->instructions) { free(p); return NULL; }
  memcpy(p->instructions, instructions, count * sizeof(uint32_t));
  p->count = count;

  if (license && license->data && license->len > 0) {
    p->license_data = (char*)malloc(license->len);
    if (p->license_data) {
      memcpy(p->license_data, license->data, license->len);
      p->license.data = p->license_data;
      p->license.len = license->len;
    }
  }

  return p;
}

void xprobe_prog_destroy(xprobe_prog* prog) {
  if (!prog) return;
  free(prog->instructions);
  free(prog->license_data);
  free(prog);
}

/* ---- Map lifecycle ---- */

xprobe_map* xprobe_map_create(const char* name, uint32_t size) {
  if (size == 0) return NULL;
  xprobe_map* m = (xprobe_map*)calloc(1, sizeof(xprobe_map));
  if (!m) return NULL;

  if (name) {
    strncpy(m->name, name, XPROBE_MAP_NAME_MAX - 1);
    m->name[XPROBE_MAP_NAME_MAX - 1] = '\0';
  }

  m->counters = (uint64_t*)calloc(size, sizeof(uint64_t));
  if (!m->counters) { free(m); return NULL; }
  m->size = size;
  return m;
}

void xprobe_map_destroy(xprobe_map* map) {
  if (!map) return;
  free(map->counters);
  free(map);
}

uint64_t xprobe_map_read(const xprobe_map* map, uint32_t index) {
  if (!map || index >= map->size) return 0;
  return map->counters[index];
}

void xprobe_map_reset(xprobe_map* map, uint32_t index) {
  if (!map || index >= map->size) return;
  map->counters[index] = 0;
}

void xprobe_map_reset_all(xprobe_map* map) {
  if (!map) return;
  memset(map->counters, 0, map->size * sizeof(uint64_t));
}

uint32_t xprobe_map_size(const xprobe_map* map) {
  return map ? map->size : 0;
}

/* ---- VM Executor ---- */

xprobe_result xprobe_exec(const xprobe_prog* prog, xprobe_map* map,
                           const xprobe_context* ctx) {
  xprobe_result result;
  result.status = XPROBE_OK;
  result.events = 0;
  result.steps = 0;

  if (!prog || !ctx) {
    result.status = XPROBE_ERR_INVALID_PROG;
    return result;
  }

  uint64_t regs[8] = {0};
  int flags = 0;
  uint32_t pc = 0;

  const uint32_t* code = prog->instructions;
  uint32_t count = prog->count;

  while (pc < count) {
    if (result.steps >= XPROBE_MAX_INSTRUCTIONS) {
      result.status = XPROBE_ERR_MAX_STEPS;
      return result;
    }

    uint32_t instr = code[pc];
    uint8_t opcode = (uint8_t)(instr >> 24);
    uint8_t dst = (uint8_t)((instr >> 21) & 0x7);
    uint8_t src = (uint8_t)((instr >> 18) & 0x7);
    uint32_t imm = instr & 0x3FFFF;

    result.steps++;
    pc++;

    switch (opcode) {
      case XPROBE_OPCODE_HALT:
        return result;

      case XPROBE_OPCODE_LOAD_IMM:
        /* sign-extend 18-bit immediate */
        regs[dst] = (uint64_t)(int64_t)(int32_t)((imm ^ 0x20000) - 0x20000);
        break;

      case XPROBE_OPCODE_LOAD_CTX:
        switch (imm & 0x7) {
          case XPROBE_CTX_USER_DATA:    regs[dst] = ctx->user_data; break;
          case XPROBE_CTX_TIMESTAMP_NS: regs[dst] = ctx->timestamp_ns; break;
          case XPROBE_CTX_SCHEMA_ID:    regs[dst] = ctx->schema_id; break;
          case XPROBE_CTX_FLAGS:        regs[dst] = ctx->flags; break;
          case XPROBE_CTX_WRITE_POS:    regs[dst] = ctx->write_pos; break;
          case XPROBE_CTX_READ_POS:     regs[dst] = ctx->read_pos; break;
          case XPROBE_CTX_CAPACITY:     regs[dst] = ctx->capacity; break;
          case XPROBE_CTX_PAYLOAD_LEN:  regs[dst] = ctx->payload_len; break;
          default: regs[dst] = 0; break;
        }
        break;

      case XPROBE_OPCODE_LOAD_MAP:
        if (map && imm < map->size) {
          regs[dst] = map->counters[imm];
        } else {
          regs[dst] = 0;
        }
        break;

      case XPROBE_OPCODE_STORE_MAP:
        if (map && imm < map->size) {
          map->counters[imm] = regs[src];
        }
        break;

      case XPROBE_OPCODE_ADD:
        regs[dst] = regs[dst] + regs[src];
        break;

      case XPROBE_OPCODE_SUB:
        regs[dst] = regs[dst] - regs[src];
        break;

      case XPROBE_OPCODE_CMP_GT:
        flags = (regs[dst] > regs[src]) ? 1 : 0;
        break;

      case XPROBE_OPCODE_JMP_IF_NOT:
        if (flags == 0) {
          pc = pc + (uint32_t)(int32_t)imm;
        }
        break;

      case XPROBE_OPCODE_INC_MAP:
        if (map && imm < map->size) {
          map->counters[imm]++;
        }
        break;

      default:
        result.status = XPROBE_ERR_INVALID_PROG;
        return result;
    }
  }

  return result;
}

/* ---- Hook management ---- */

static xprobe_handle* g_hooks[XPROBE_HOOK_COUNT] = {NULL};

xprobe_handle* xprobe_attach(xprobe_prog* prog, xprobe_map* map, xprobe_hook hook) {
  if (!prog || hook >= XPROBE_HOOK_COUNT) return NULL;

  xprobe_handle* h = (xprobe_handle*)calloc(1, sizeof(xprobe_handle));
  if (!h) return NULL;
  h->prog = prog;
  h->map = map;
  h->hook = hook;

  /* Replace existing hook at this slot */
  xprobe_handle* old = g_hooks[hook];
  g_hooks[hook] = h;
  free(old);

  return h;
}

void xprobe_detach(xprobe_handle* handle) {
  if (!handle) return;
  if (handle->hook < XPROBE_HOOK_COUNT && g_hooks[handle->hook] == handle) {
    g_hooks[handle->hook] = NULL;
  }
  free(handle);
}

/* Internal: get hook for a given type (used by channel integration) */
xprobe_handle* xprobe_get_hook(xprobe_hook hook) {
  if (hook >= XPROBE_HOOK_COUNT) return NULL;
  return g_hooks[hook];
}
```
